func TestManagementKeeperNamesS1ReadsIdentity(t *testing.T) {
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        if r.URL.Path != "/api/v1/usage/identities" { http.NotFound(w,r); return }
        w.Header().Set("Content-Type","application/json")
        _, _ = w.Write([]byte("{\"identities\":[{\"id\":\"17\",\"identity\":\"idx-a\",\"auth_type\":1,\"type\":\"codex\",\"provider\":\"codex\",\"alias\":\"生产\",\"displayName\":\"生产\",\"is_deleted\":false}]}"))
    }))
    defer server.Close()
    setupManagementTest(t, Config{Enabled:true, UsageKeeperURL:server.URL})
    got := managementKeeperAuthNames(false)
    if got.StatusCode != 200 { t.Fatalf("status=%d",got.StatusCode) }
    var body struct { Status string; Items []struct { AuthIndex string `json:"auth_index"`; DisplayName string `json:"display_name"` } }
    decodeBody(t,got,&body)
    if body.Status!="ready" || len(body.Items)!=1 || body.Items[0].AuthIndex!="idx-a" || body.Items[0].DisplayName!="生产" {
        t.Fatalf("unexpected response: %s",got.Body)
    }
}
~~~
