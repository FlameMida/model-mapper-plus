func TestManagementKeeperNamesS3ImmediateUpdate(t *testing.T) {
    var mu sync.Mutex
    alias := "原名称"
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request) {
        mu.Lock(); defer mu.Unlock()
        if r.Method==http.MethodPatch && r.URL.Path=="/api/v1/usage/identities/17" {
            if r.Header.Get("X-CPA-Usage-Keeper-Request")!="fetch" { w.WriteHeader(403); return }
            var body struct{ Alias string }
            if json.NewDecoder(r.Body).Decode(&body)!=nil { w.WriteHeader(400); return }
            alias=body.Alias
        } else if r.Method!=http.MethodGet || r.URL.Path!="/api/v1/usage/identities" {
            http.NotFound(w,r); return
        }
        row:=map[string]any{"id":"17","identity":"idx-a","auth_type":1,"type":"codex",
            "provider":"codex","alias":alias,"displayName":alias,"is_deleted":false}
        w.Header().Set("Content-Type","application/json")
        if r.Method==http.MethodGet {
            _=json.NewEncoder(w).Encode(map[string]any{"identities":[]any{row}})
        } else { _=json.NewEncoder(w).Encode(row) }
    }))
    defer server.Close()
    setupManagementTest(t,Config{Enabled:true,UsageKeeperURL:server.URL})
    got:=dispatchManagement(pluginapi.ManagementRequest{Method:http.MethodPatch,
        Path:managementHandleBase+"/keeper/auth-names",
        Body:[]byte("{\"auth_index\":\"idx-a\",\"alias\":\"新名称\"}")})
    if got.StatusCode!=200 { t.Fatalf("status=%d body=%s",got.StatusCode,got.Body) }
    var body struct{ Status string }
    decodeBody(t,got,&body)
    mu.Lock(); saved:=alias; mu.Unlock()
    if body.Status!="ready" || saved!="新名称" { t.Fatalf("response=%s stored=%q",got.Body,saved) }
}
~~~

**步骤 3：最小实现**

management.go登记并挂载T03 GET/POST名称读取handler，再新增PATCH分支直接调用managementPatchKeeperAuthName；该handler内部调用withAuditedManagement一次，其run闭包返回业务response与显式auditResult，禁止dispatch再包一次。审计Begin先落盘；客户端传Keeper身份ID时忽略，必须对auth_index重新读取当前列表后解析ID。预检和PATCH共用5秒ctx。
业务状态映射关键函数：
~~~go
func keeperNameAuditOutcome(status string) string {
    switch status {
    case "ready": return "succeeded"
    case "unknown": return "unknown"
    default: return "failed"
    }
}
~~~
输入非法→HTTP200/invalid；未匹配→not_found；预检/登录失败→unavailable；实际PATCH已发出却无法确认→unknown（Changed=nil）；ready采用返回item并使旧名称缓存失效，下一次读取不覆盖为旧在途结果。
首次401可登录后重试一次（总期不重置），5xx/timeout不重试PATCH。Cookie和密码不返回浏览器。预检失败不可误标unknown；不能只凭HTTP200设置审计成功。
明确处理双失败：PATCH未知+Finish失败→status unknown、audit.recorded=false；前端不得显示已保存/已回滚。日志Begin失败时httptest PATCH次数为0。
新增channel阻塞的fake server场景验证网络不持配置/state锁，管理写互斥期间不死锁；写前后名称均从可信响应投影取得并脱敏。
