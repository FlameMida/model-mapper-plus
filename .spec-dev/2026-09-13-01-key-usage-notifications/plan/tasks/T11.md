`NotificationScheduleEditor.tsx`：

```tsx
import { InputNumber, Select, TimePicker } from '@douyinfe/semi-ui'
import type { NotificationSchedule, ScheduleKind } from '../notifications'

export default function NotificationScheduleEditor({ value, onChange }: {
  value: NotificationSchedule
  onChange: (s: NotificationSchedule) => void
}) {
  const set = (patch: Partial<NotificationSchedule>) => onChange({ ...value, ...patch })
  return (
    <div style={{ display: 'flex', gap: 8, alignItems: 'center', flexWrap: 'wrap' }}>
      <Select value={value.kind} onChange={(k: ScheduleKind) => onChange({ kind: k, time: k === 'interval' ? '' : value.time })} style={{ width: 110 }}
        optionList={[{ value: 'interval', label: '每隔' }, { value: 'monthly', label: '每月' }, { value: 'yearly', label: '每年' }]} />
      {value.kind === 'interval' && (
        <>
          <InputNumber value={value.interval ?? 1} min={1} onChange={(v) => set({ interval: Number(v) })} style={{ width: 80 }} />
          {/* 单位 Select（秒/分/时/天）仅改 interval 数值：value.interval = n * unitSeconds */}
          <UnitSelect onUnit={(sec) => set({ interval: Math.max(1, Math.round((value.interval ?? 1) * sec / baseUnit(value.interval ?? 1))) })} />
        </>
      )}
      {value.kind === 'monthly' && (
        <Select value={value.month_end ? 'end' : 'start'} onChange={(v) => set({ month_end: v === 'end' })} style={{ width: 100 }}
          optionList={[{ value: 'start', label: '月初' }, { value: 'end', label: '月末' }]} />
      )}
      {value.kind === 'yearly' && (
        <>
          <Select value={value.month ?? 1} onChange={(m) => set({ month: Number(m) })} style={{ width: 90 }}
            optionList={Array.from({ length: 12 }, (_, i) => ({ value: i + 1, label: `${i + 1}` }))} />
          <Select value={value.day ?? 1} onChange={(d) => set({ day: Number(d) })} style={{ width: 90 }}
            optionList={Array.from({ length: 31 }, (_, i) => ({ value: i + 1, label: `${i + 1}` }))} />
        </>
      )}
      {value.kind !== 'interval' && (
        <TimePicker format="HH:mm:ss" value={value.time} onChange={(t) => set({ time: t ?? '' })} />
      )}
    </div>
  )
}
```

（`UnitSelect`/`baseUnit` 为文件内 10 行辅助：单位秒/分/时/天 ↔ 1/60/3600/86400，切换时按当前数值折算；间隔校验沿用 Go 侧 `1..31*86400`，超界 UI 即时提示。）
