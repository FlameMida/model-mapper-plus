**步骤 3：写最小实现**

`KeyNotificationsTab.tsx` 结构：

```tsx
import { Button, Table, Switch, Typography, Modal } from '@douyinfe/semi-ui'
import { useState } from 'react'
import { api, type KeyBinding } from '../api'
import type { Notification, DeliveryRecord } from '../notifications'
import NotificationEditor from '../components/NotificationEditor'

export default function KeyNotificationsTab({ binding, siblingNames, onChange }: {
  binding: KeyBinding
  siblingNames: string[]
  onChange: (notifications: Notification[]) => void
  globalName?: string
  globalSchedule?: string
}) {
  const list = binding.notifications ?? []
  const [editing, setEditing] = useState<{ draft: Notification; originalName: string } | null>(null)
  const [expanded, setExpanded] = useState<string>()

  const toggle = (n: Notification, enabled: boolean) =>
    onChange(list.map((x) => (x.id === n.id ? { ...x, enabled } : x)))

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
      <div style={{ display: 'flex' }}>
        <Button theme="solid" style={{ marginLeft: 'auto' }}
          onClick={() => setEditing({ draft: newDraft(), originalName: '' })}>＋ 新增通知</Button>
      </div>
      {list.length === 0 ? (
        <div style={{ border: '1px dashed var(--semi-color-border)', borderRadius: 8, padding: 20, textAlign: 'center' }}>
          <Typography.Paragraph>该 Key 尚未配置专属通知</Typography.Paragraph>
          <Typography.Text type="tertiary">
            当前按全局默认通知「{globalName ?? '用量通知'}」发送（{globalSchedule ?? '跟随全局计划'}）——新增第一条专属通知后，全局默认通知即对本 Key 停止。
          </Typography.Text>
        </div>
      ) : (
        <Table dataSource={list} rowKey="id" pagination={false}
          expandedRowKeys={expanded ? [expanded] : []}
          onExpandedRowsChange={(keys) => setExpanded(keys[0] as string | undefined)}
          expandedRowRender={(n?: Notification) => <DeliveriesTable notificationId={n!.id} binding={binding} />}
          columns={[
            { title: '通知名称', dataIndex: 'name' },
            { title: '启用', dataIndex: 'enabled', render: (_v, n) => (
                <Switch aria-label={`启用通知：${n.name}`} checked={n.enabled} onChange={(v) => toggle(n, v)} />) },
            { title: '发送计划', render: (_v, n) => <ScheduleSummary n={n} /> },
            { title: '平台', render: (_v, n) => (n.platforms ?? []).filter((p) => p.enabled).map((p) => p.kind).join('、') || '—' },
            { title: '最近投递', render: (_v, n) => <LastDelivery cell binding={binding} notificationId={n.id} /> },
            { title: '操作', render: (_v, n) => (
                <>
                  <Button size="small" style={{ marginRight: 4 }} onClick={() => setEditing({ draft: n, originalName: n.name })}>编辑</Button>
                  <Button size="small" style={{ marginRight: 4 }}
                    onClick={() => api.notifications.testSend({ key: binding.key, notification_id: n.id })
                      .then(() => Toast.success('测试发送已受理'))
                      .catch((e: Error) => Toast.error(e.message))}>测试发送</Button>
                  <Button size="small" type="danger" onClick={() => Modal.confirm({
                    title: '删除通知', content: `确认删除 ${n.name}？`,
                    onOk: () => onChange(list.filter((x) => x.id !== n.id)),
                  })}>删除</Button>
                </>) },
          ]} />
      )}
      {editing && (
        <NotificationEditor visible originalName={editing.originalName}
          siblingNames={list.map((n) => n.name).filter((n) => n !== editing.originalName)}
          initial={editing.draft}
          onSaved={(n) => onChange(editing.originalName
            ? list.map((x) => (x.id === n.id ? n : x))
            : [...list, n])}
          onClose={() => setEditing(null)} />
      )}
    </div>
  )
}
```

`newDraft()`：`{ id: crypto.randomUUID(), name: '', enabled: true, template_follows_global: true, schedule_follows_global: true, modules: [], platforms: [{kind:'wecom',enabled:false},{kind:'feishu',enabled:false},{kind:'dingtalk',enabled:false}] }`（新建默认跟随全局——2026-09-13 确认）。`ScheduleSummary`/`LastDelivery`/`DeliveriesTable` 为文件内小函数：计划摘要（每隔 N 秒/月初|月末 HH:mm:ss/每年 M 月 D 日 HH:mm:ss）、最近投递（`api.notifications.deliveries({ key_fingerprint, notification_id, limit: 1 })`，accepted 绿 / failed 红 + error_code / unknown 灰）、展开表（`deliveries({ key_fingerprint, notification_id, limit: 20 })` + 失败行「重试」按钮调 `retryDelivery`）。`Toast` 从 semi-ui 引入；测试环境 `Modal.confirm` 若阻塞，沿用 `KeysPanel.delete.test.tsx` 的既有 mock 方式（文件里已有该模式）。

- 顶部 import `KeyNotificationsTab`；`normalizeBinding` 追加 `notifications: binding.notifications ?? []`。
