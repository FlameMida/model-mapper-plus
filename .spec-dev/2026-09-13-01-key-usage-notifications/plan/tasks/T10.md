**步骤 3：写最小实现**

`web/src/notifications.ts`：

```ts
export type PlatformKind = 'wecom' | 'feishu' | 'dingtalk'
export type ModuleKind = 'daily' | 'monthly' | 'half_year' | 'yearly' | 'window_5h' | 'weekly' | 'reset_cards'
export type PeriodKind = 'current' | 'previous'
export type ScheduleKind = 'interval' | 'monthly' | 'yearly'
export const STAT_MODULE_KINDS: ModuleKind[] = ['daily', 'monthly', 'half_year', 'yearly']
export const WINDOW_MODULE_KINDS: ModuleKind[] = ['window_5h', 'weekly', 'reset_cards']
export const MODULE_KINDS = [...STAT_MODULE_KINDS, ...WINDOW_MODULE_KINDS]
export const MODULE_LABELS: Record<ModuleKind, string> = {
  daily: '日统计', monthly: '月统计', half_year: '半年统计', yearly: '年统计',
  window_5h: '5H 窗口', weekly: 'Weekly 窗口', reset_cards: '重置卡',
}

export interface ModuleConfig { kind: ModuleKind; period?: PeriodKind }
export interface NotificationSchedule {
  kind: ScheduleKind
  interval?: number
  month_end?: boolean
  month?: number
  day?: number
  time: string
}
export interface PlatformIdentity {
  kind: PlatformKind
  enabled: boolean
  webhook?: string
  user_ids?: string[]
  sign_secret?: string
}
export interface Notification {
  id: string
  name: string
  enabled: boolean
  template_follows_global?: boolean
  schedule_follows_global?: boolean
  modules?: ModuleConfig[]
  schedule?: NotificationSchedule | null
  platforms?: PlatformIdentity[]
}
export interface NotificationSettings { enabled: boolean; global_default: Notification }
export interface NotificationStatus {
  running: boolean
  error_code?: string
  revision: number
  pending_jobs: number
  next_fire?: string
}
export interface DeliveryRecord {
  id: string
  job_id: string
  key_fingerprint: string
  notification_id: string
  platform: PlatformKind
  period_key: string
  outcome: 'accepted' | 'failed' | 'unknown'
  error_code?: string
  detail?: string
  created_at: string
}
export interface PreviewResponse { text: string; warnings: string[]; bytes: number }
```

`api.ts` 的 `api` 对象追加（复用现有 `call`，自动获得 401 跳转与 `unescapeDeep`）：

```ts
  notifications: {
    getSettings: () => call<NotificationSettings>('GET', '/notifications/settings'),
    putSettings: (s: NotificationSettings) => call<NotificationSettings>('PUT', '/notifications/settings', s),
    getStatus: () => call<NotificationStatus>('GET', '/notifications/status'),
    preview: (req: { key?: string; notification_id?: string }) =>
      call<PreviewResponse>('POST', '/notifications/preview', req),
    testSend: (req: { key?: string; notification_id?: string }) =>
      call<{ delivery_ids: string[] }>('POST', '/notifications/test-send', req),
    deliveries: (filter: { key_fingerprint?: string; notification_id?: string; platform?: string; outcome?: string; limit?: number }) => {
      const q = new URLSearchParams()
      for (const [k, v] of Object.entries(filter)) if (v) q.set(k, String(v))
      return call<{ items: DeliveryRecord[] }>('GET', `/notifications/deliveries${q.size ? `?${q}` : ''}`)
    },
    retryDelivery: (id: string) => call<{ job_id: string }>('POST', '/notifications/deliveries/retry', { id }),
  },
```

顶部补 `import type { ... } from './notifications'`。
