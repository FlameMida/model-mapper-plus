import { fireEvent, render, screen, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { describe, expect, it, vi } from 'vitest'
import NotificationEditor from './NotificationEditor'
import type { Notification } from '../notifications'

const draft: Notification = {
  id: 'n1', name: '日报用量', enabled: true,
  template_follows_global: false, schedule_follows_global: false,
  modules: [{ kind: 'daily', period: 'current' }],
  schedule: { kind: 'interval', interval: 86400, time: '' },
  platforms: [{ kind: 'feishu', enabled: true, webhook: 'https://f', user_ids: ['ou_a'] }],

function renderEditor(over: Partial<Notification> = {}, names: string[] = []) {
  const onSaved = vi.fn()
  render(<NotificationEditor visible originalName="日报用量" siblingNames={names}
    initial={{ ...draft, ...over }} onSaved={onSaved} onClose={() => {}} />)
  return { onSaved }
}

describe('NotificationEditor', () => {
  it('blocks save when name collides with a sibling', async () => {
    const { onSaved } = renderEditor({ name: '周报用量' }, ['周报用量'])

    expect(await screen.findByText(/同 Key 内已存在同名通知/)).toBeInTheDocument()
    expect(onSaved).not.toHaveBeenCalled()
  })

  it('blocks save when enabled platform lacks identity', async () => {
    const { onSaved } = renderEditor({
      platforms: [{ kind: 'feishu', enabled: true, webhook: 'https://f' }],
    })
    await userEvent.click(screen.getByRole('button', { name: '保存' }))
    expect(await screen.findByText(/用户唯一 ID/)).toBeInTheDocument()
    expect(onSaved).not.toHaveBeenCalled()
  })

  it('shows stale-preview banner after edit and previews saved entity', async () => {

    vi.doMock('../api', () => ({ api: { notifications: { preview } } }))

**步骤 3：写最小实现**

结构（完整组件 ~180 行，关键决策如下）：

```tsx
import { SideSheet, Button, Tabs, TabPane, Input, Switch, Banner, Typography } from '@douyinfe/semi-ui'
import { useEffect, useMemo, useState } from 'react'
import { api } from '../api'
import type { Notification } from '../notifications'
import NotificationModulesEditor from './NotificationModulesEditor'
import NotificationScheduleEditor from './NotificationScheduleEditor'
import PlatformIdentityEditor, { validatePlatforms } from './PlatformIdentityEditor'

export default function NotificationEditor(props: /* 见接口块 */) {
  const [draft, setDraft] = useState<Notification>(props.initial)
  const [dirty, setDirty] = useState(false)
  const [preview, setPreview] = useState<{ text: string } | null>(null)
  const [error, setError] = useState('')
  // initial 变化（切换编辑对象）时重置草稿
  useEffect(() => { setDraft(props.initial); setDirty(false); setPreview(null) }, [props.initial])
  const edit = (patch: Partial<Notification>) => { setDraft((d) => ({ ...d, ...patch })); setDirty(true) }

  const saveBlocked = useMemo(() => {
    const name = draft.name.trim()
    if (!name) return '通知名称为必填项'
    const others = props.siblingNames.filter((n) => props.originalName === '' || n !== props.originalName)
    if (others.includes(name)) return '同 Key 内已存在同名通知'
    if (!draft.template_follows_global && !(draft.modules?.length)) return '至少启用一个统计模块'
    const errs = validatePlatforms(draft.platforms ?? [])
    return errs[0] ?? ''
  }, [draft, props.siblingNames, props.originalName])

  return (
    <SideSheet title={`编辑通知：${props.originalName || draft.name}`} visible={props.visible}
      onCancel={props.onClose} size="medium" keepDOM
      footer={
        <div style={{ display: 'flex', gap: 8, justifyContent: 'flex-end' }}>
          {preview && <Typography.Text size="small" type="tertiary" style={{ marginRight: 'auto' }}>预览：{preview.text.slice(0, 80)}…</Typography.Text>}
          <Button onClick={doPreview}>预览消息</Button>
          <Button onClick={props.onClose}>取消</Button>
          <Button theme="solid" disabled={!!saveBlocked} onClick={doSave}>保存</Button>
        </div>
      }>
      <Tabs type="line">
        <TabPane tab="通知模板" itemKey="template">
          <div style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
            <Input aria-label="通知名称" value={draft.name}
              onChange={(name) => edit({ name })} />
            {saveBlocked.startsWith('同 Key') && <Typography.Text type="danger">{saveBlocked}</Typography.Text>}
            <label style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
              <Switch checked={draft.enabled} onChange={(enabled) => edit({ enabled })} />
              <Typography.Text>启用通知（关闭只停发，不删除配置）</Typography.Text>
            </label>
            <label style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
              <Switch checked={!!draft.template_follows_global}
                onChange={(template_follows_global) => edit({ template_follows_global })} />
              <Typography.Text>跟随全局默认模板</Typography.Text>
            </label>
            {draft.template_follows_global
              ? <Typography.Text type="tertiary">模块序列与周期将使用全局默认模板（来源：全局默认通知）</Typography.Text>
              : <NotificationModulesEditor value={draft.modules ?? []} onChange={(modules) => edit({ modules })} />}
            <label style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
              <Switch checked={!!draft.schedule_follows_global}
                onChange={(schedule_follows_global) => edit({ schedule_follows_global })} />
              <Typography.Text>跟随全局默认计划</Typography.Text>
            </label>
            {!draft.schedule_follows_global && draft.schedule &&
              <NotificationScheduleEditor value={draft.schedule} onChange={(schedule) => edit({ schedule })} />}
          </div>
        </TabPane>
        <TabPane tab="平台身份配置" itemKey="platforms">
          <PlatformIdentityEditor value={draft.platforms ?? []} onChange={(platforms) => edit({ platforms })} />
        </TabPane>
      </Tabs>
      {dirty && <Banner type="warning" bordered description="配置自上次预览后已修改——预览结果可能过期，发送前请重新预览。" style={{ marginTop: 12 }} />}
      {error && <Banner type="danger" bordered description={error} />}
    </SideSheet>
  )

  async function doPreview() {
    try {
      const r = await api.notifications.preview({ notification_id: draft.id || undefined })
      setPreview({ text: r.text }); setDirty(false); setError('')
    } catch (e) { setError((e as Error).message) }
  }
  function doSave() {
    if (saveBlocked) return
    props.onSaved({ ...draft, name: draft.name.trim() })
    props.onClose()
  }
}
```

保存本身不在此发请求：T14/T15 把 `onSaved` 的草稿并入 binding/全局实体再走 `postKey`/`putSettings`（与既有「编辑绑定→Modal onOk→postKey」保存路径一致）。test-send 在 T14 列表行内（针对已保存通知），抽屉不重复。
